"""Pre-registered A-background range probes for the cheap R2/R1 variants."""
from concurrent.futures import ThreadPoolExecutor
import json
import os
from pathlib import Path
import subprocess
import sys

BASE = Path(__file__).resolve().parent
REPO = BASE / "rs-range-screen"
DATA = BASE / "resense-p3-data"
OUT = BASE / "resense-p3-runs/range-screen-probes"
BAG = "roundT_squareT_pressureGate_squareT"


def run(job):
    name, obj, settings = job
    prefix = OUT / name
    env = dict(os.environ, PYTHONPATH=os.pathsep.join((str(REPO), str(BASE))), RESENSE_REPO=str(REPO), WT=str(REPO),
               OMP_NUM_THREADS="1", OPENBLAS_NUM_THREADS="1", MKL_NUM_THREADS="1", DARKBAND="0", CFG=",".join(settings))
    cmd = [sys.executable, str(BASE / "resense-p3-run-script.py"), str(REPO),
           str(REPO / "docs/evidence/judgement_2026-10-13/scripts/range_test.py"),
           str(DATA / "decoded" / BAG), str(DATA / "odometry" / f"icp_{BAG}.json"), str(prefix),
           "--obj", obj, "--start", "150", "--d0", "230", "--end", "340"]
    with open(str(prefix) + ".log", "w") as log:
        result = subprocess.run(cmd, cwd=REPO, env=env, stdout=log, stderr=subprocess.STDOUT)
    print(name, result.returncode, flush=True)
    if result.returncode:
        raise RuntimeError(name)
    report = json.loads(Path(str(prefix) + ".json").read_text())
    return name, {"set": settings, "summary": report["summary"]}


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    protocol = json.loads((REPO / "docs/evidence/range_screen_2026-10-15/protocol.json").read_text())
    jobs = [("baseline_cube05", "0.5,0.5,0.5", []), ("baseline_cube03", "0.3,0.3,0.3", [])]
    jobs += [(v["id"], "0.5,0.5,0.5" if v["id"].startswith("R2") else "0.3,0.3,0.3", v["set"])
             for v in protocol["variants_in_order"]]
    with ThreadPoolExecutor(max_workers=2) as pool:
        result = dict(pool.map(run, jobs))
    (OUT / "summary.json").write_text(json.dumps(result, indent=2), encoding="utf-8")
    print("RANGE SCREEN COMPLETE", flush=True)


if __name__ == "__main__":
    main()
