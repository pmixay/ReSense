"""Run the baseline and H1/H2 gates only after complete verified intake."""
import json
import os
from pathlib import Path
import subprocess
import sys
import time

BASE = Path(__file__).resolve().parent
DATA = BASE / "resense-p3-data"
OUT = BASE / "resense-p3-runs/gates"
JOBS = [("baseline", BASE / "rs-range-screen"), ("H1", BASE / "rs-h1"), ("H2", BASE / "rs-h2")]


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    required = [DATA / "cache/new_data/intake_manifest.json", DATA / "cache/cloud_with_fake_obj/cloud_with_fake_obj_stamps.json",
                DATA / "decoded/squareT_platform_squareT_switch/meta.json"]
    while not all(path.exists() for path in required):
        time.sleep(10)
    exits = {}
    for tag, repo in JOBS:
        env = dict(os.environ, PYTHONPATH=os.pathsep.join((str(repo), str(BASE))), RESENSE_REPO=str(repo),
                   OMP_NUM_THREADS="1", OPENBLAS_NUM_THREADS="1", MKL_NUM_THREADS="1")
        cmd = [sys.executable, "scripts/regression_gate.py", "--cache", str(DATA / "cache"), "--jobs", "3",
               "--baseline", "docs/evidence/results/regression_baseline_2026-09-29_shell_off.json",
               "--out", str(OUT / f"{tag}.json"), "--work", str(OUT / f"{tag}_work")]
        print(time.strftime("%H:%M:%S"), tag, "START", flush=True)
        with open(OUT / f"{tag}.log", "w") as log:
            result = subprocess.run(cmd, cwd=repo, env=env, stdout=log, stderr=subprocess.STDOUT)
        exits[tag] = result.returncode
        (OUT / "exits.json").write_text(json.dumps(exits, indent=2), encoding="utf-8")
        print(time.strftime("%H:%M:%S"), tag, "EXIT", result.returncode, flush=True)
        if tag == "baseline" and result.returncode:
            raise RuntimeError("baseline did not reproduce; do not label candidate differences as regressions yet")
    print("GATE RUNS FINISHED", flush=True)


if __name__ == "__main__":
    main()
