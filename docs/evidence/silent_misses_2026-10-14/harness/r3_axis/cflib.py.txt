"""Single-frame counterfactuals of the wall curvature on obs records (scratch): sealed average vs
(i) the right/left side cut at the first gap > G m from the near field, (ii) equal weights, (iii) the
side whose near-field-continuous run reaches farthest. Prints raw curvature per variant and the true
curvature proxy (left-wall-based true axis second derivative at 60-100 m)."""
import json, sys
import numpy as np


def fit_side(x, y, t, maxres=0.4, minb=6, maxrms=0.35):
    def quad(xb, yb):
        A = np.stack([np.ones_like(xb), 0.5 * xb * xb], 1)
        s, *_ = np.linalg.lstsq(A, yb - t * xb, rcond=None)
        return s[0], s[1]
    if x.size < minb:
        return None
    a, k = quad(x, y)
    xb, yb = x, y
    for _ in range(2):
        res = yb - (a + t * xb + 0.5 * k * xb * xb)
        keep = np.abs(res) < maxres
        if keep.sum() < minb:
            return None
        xb, yb = xb[keep], yb[keep]
        a, k = quad(xb, yb)
    res = yb - (a + t * xb + 0.5 * k * xb * xb)
    rms = float(np.sqrt(np.mean(res ** 2)))
    if rms > maxrms:
        return None
    return k, rms, float(xb.max())


def cut_gap(x, y, G):
    keep = [0]
    for i in range(1, x.size):
        if x[i] - x[keep[-1]] > G:
            break
        keep.append(i)
    return x[keep], y[keep]


def combine(fits, eq=False):
    fits = [f for f in fits if f is not None]
    if not fits:
        return None
    w = np.array([1.0 if eq else 1.0 / (f[1] ** 2 + 1e-4) for f in fits])
    return float(np.sum(w * [f[0] for f in fits]) / w.sum())

def follow(x, y, c, t, k):
    x = np.asarray(x, float); y = np.asarray(y, float)
    dy = y - (c + t * x + 0.5 * k * x * x)
    near = x <= 30.0
    if near.sum() < 3:
        return None
    a = dy * np.sign(np.median(dy[near]))
    ref = float(np.median(a[near]))
    if not (0 < ref <= 3.5) or float(np.median(np.abs(a[near] - ref))) > 0.25 or abs(np.polyfit(x[near], a[near], 1)[0]) > 0.02:
        return None
    out = []
    prev, prev_a = float(x[near][-1]), float(a[near][-1])
    for xi, ai in zip(x[~near], a[~near]):
        if xi - prev > 12.0:
            break
        if abs(ai - prev_a) > 2e-3 * xi * (xi - prev):
            ref += ai - prev_a
        prev, prev_a = float(xi), float(ai)
        out.append((xi, ref - ai))
    return out


def classify(curv, i):
    lo, hi = max(0, i - 5), min(len(curv) - 1, i + 5)
    if abs(curv[hi] - curv[lo]) > 2e-4:
        return "entry/exit"
    return "curve" if abs(curv[i]) > 3e-4 else "straight"
