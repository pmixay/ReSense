"""Monitored range / clear_distance of a gate run vs the sealed gate, per recording and ride (scratch).
usage: gcost.py <work_dir>"""
import glob, json, os, sys
import numpy as np
new_dir = sys.argv[1]
old_dir = "/tmp/claude-0/fixwork/gate_head/work"
ride = {"n": 0, "c": 0, "mon_lost": 0.0, "mon_tot": 0.0, "cl_lost": 0.0, "cl_tot": 0.0, "caps": [], "mon_new": [], "mon_old": []}
for f in sorted(glob.glob(os.path.join(new_dir, "*.jsonl"))):
    name = os.path.basename(f)
    a = [json.loads(l) for l in open(f)]
    b = [json.loads(l) for l in open(os.path.join(old_dir, name))]
    n = len(a)
    c = sum(1 for x, y in zip(a, b) if x["health"]["monitored_range"] < y["health"]["monitored_range"])
    ml = sum(y["health"]["monitored_range"] - x["health"]["monitored_range"] for x, y in zip(a, b))
    mt = sum(y["health"]["monitored_range"] for y in b)
    cl = sum(y["clear_distance"] - x["clear_distance"] for x, y in zip(a, b))
    ct = sum(y["clear_distance"] for y in b)
    caps = [x["health"]["wall_drift"] for x, y in zip(a, b) if "wall_drift" in x["health"] and x["health"]["monitored_range"] < y["health"]["monitored_range"]]
    mn = [x["health"]["monitored_range"] for x in a]
    mo = [y["health"]["monitored_range"] for y in b]
    q = lambda v, p: round(float(np.percentile(v, p)), 1) if v else None
    print("%-42s frames %5d capped %5d (%5.1f%%) cap q10/50/90 %s/%s/%s  monitored lost %4.1f%%  clear lost %4.1f%%  monitored median %s -> %s" % (
        name, n, c, 100.0 * c / n, q(caps, 10), q(caps, 50), q(caps, 90), 100 * ml / mt, 100 * cl / ct, q(mo, 50), q(mn, 50)))
    if name.startswith("new_data"):
        ride["n"] += n; ride["c"] += c; ride["mon_lost"] += ml; ride["mon_tot"] += mt; ride["cl_lost"] += cl; ride["cl_tot"] += ct
        ride["caps"] += caps; ride["mon_new"] += mn; ride["mon_old"] += mo
q = lambda v, p: round(float(np.percentile(v, p)), 1) if v else None
print("RIDE frames %d capped %d (%.1f%%) cap q10/50/90 %s/%s/%s monitored lost %.1f%% clear lost %.1f%% monitored q10/50 %s/%s -> %s/%s" % (
    ride["n"], ride["c"], 100.0 * ride["c"] / ride["n"], q(ride["caps"], 10), q(ride["caps"], 50), q(ride["caps"], 90),
    100 * ride["mon_lost"] / ride["mon_tot"], 100 * ride["cl_lost"] / ride["cl_tot"], q(ride["mon_old"], 10), q(ride["mon_old"], 50),
    q(ride["mon_new"], 10), q(ride["mon_new"], 50)))
