"""Cross-section scatter of dumped points (scratch): Y (or dy from a given axis) vs h for X slabs.
usage: xsec.py <pts.npz> k x0,x1[;x0,x1...] out.png"""
import sys
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
P = np.load(sys.argv[1])["k%s" % sys.argv[2]]
slabs = [tuple(float(v) for v in s.split(",")) for s in sys.argv[3].split(";")]
fig, axs = plt.subplots(1, len(slabs), figsize=(4 * len(slabs), 4), squeeze=False)
for ax, (x0, x1) in zip(axs[0], slabs):
    m = (P[:, 0] >= x0) & (P[:, 0] < x1)
    ax.scatter(P[m, 1], P[m, 2], s=1)
    ax.set_title("%g-%g m n=%d" % (x0, x1, m.sum()))
    ax.grid(True)
plt.tight_layout()
plt.savefig(sys.argv[4], dpi=80)
