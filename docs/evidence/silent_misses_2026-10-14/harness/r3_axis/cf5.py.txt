"""Wall-referenced error proxy of single-frame raw axes: sealed weighting vs a B variant (scratch).
For each frame with the rails' tangent and two fitted sides: raw curvature sealed / variant; each axis
(center from the record, tangent t_fixed) evaluated with ride_err.follow on both sides; per band the median
|drift| over frames, by class.
usage: MODE=w|fit cf5.py G P files..."""
import json, os, sys
import numpy as np
from cflib import fit_side, cut_gap
from cflib import follow, BANDS

G, P = float(sys.argv[1]), float(sys.argv[2])
MODE = os.environ.get("MODE", "w")


def comb(f, p):
    f = [v for v in f if v is not None]
    if not f:
        return None
    w = np.array([1.0 / (v[1] ** 2 + 1e-4) for v in f])
    if p > 0 and len(f) == 2:
        reach = np.array([max(v[2] - 30.0, 4.0) for v in f])
        w = w * (reach / reach.max()) ** p
    return float(np.sum(w * [v[0] for v in f]) / w.sum())


res = {}
for path in sys.argv[3:]:
    rows = [json.loads(l) for l in open(path)]
    curv = [r["curv"] for r in rows]
    for i, r in enumerate(rows):
        t = r["tf"]
        if t is None or len(r["sides"]) < 2:
            continue
        S = [(np.array(x), np.array(y)) for x, y, ok in r["sides"]]
        base = [fit_side(x, y, t) for x, y in S]
        if MODE == "w":
            var = [None if b is None else (b[0], b[1], float(cut_gap(x, y, G)[0].max())) for b, (x, y) in zip(base, S)]
        else:
            var = [fit_side(*(cut_gap(x, y, G) if G > 0 else (x, y)), t) for x, y in S]
        kb, kv = comb(base, 0), comb(var, P)
        if kb is None or kv is None:
            continue
        lo, hi = max(0, i - 5), min(len(curv) - 1, i + 5)
        cls = "entry/exit" if abs(curv[hi] - curv[lo]) > 2e-4 else ("curve" if abs(curv[i]) > 3e-4 else "straight")
        for name, k in (("sealed", kb), ("variant", kv)):
            best = {}
            for x, y in S:
                f = follow(x, y, r["c"], t, k)
                for xi, d in (f or []):
                    for b in BANDS:
                        if b[0] <= xi < b[1] and abs(d) > abs(best.get(b, 0.0)):
                            best[b] = d
            for b in BANDS:
                if b in best:
                    res.setdefault((cls, name, b), []).append(abs(best[b]))
for cls in ("straight", "curve", "entry/exit"):
    print(cls)
    for b in BANDS:
        s = res.get((cls, "sealed", b), [])
        v = res.get((cls, "variant", b), [])
        if s and v:
            print("   %3d-%3d m  sealed median %.2f p90 %.2f | variant median %.2f p90 %.2f  (n %d/%d)" % (
                b[0], b[1], np.median(s), np.percentile(s, 90), np.median(v), np.percentile(v, 90), len(s), len(v)))
