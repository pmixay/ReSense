"""Wall-referenced axis error proxy on gobs records (scratch): per frame and side, the near-wall offset change
(reference minus offset, + = the wall came nearer the axis = the axis is off towards it) at the followed bins
(same reference / gap / step rules as track.wall_drift_range); per frame the largest |change| per distance band.
Frames are classed by the curvature trend over +-5 frames: 'entry/exit' (|d curv| > 2e-4 over 10 frames),
'curve' (|curv| > 3e-4, steady), 'straight'.
usage: ride_err.py <gobs.jsonl> [...]"""
import json, sys
import numpy as np

BANDS = [(40, 60), (60, 80), (80, 100), (100, 120), (120, 140)]


def follow(x, y, c, t, k):
    x = np.asarray(x, float); y = np.asarray(y, float)
    dy = y - (c + t * x + 0.5 * k * x * x)
    near = x <= 30.0
    if near.sum() < 3:
        return None
    a = dy * np.sign(np.median(dy[near]))
    ref = float(np.median(a[near]))
    if not (0 < ref <= 3.5) or float(np.median(np.abs(a[near] - ref))) > 0.25 or abs(np.polyfit(x[near], a[near], 1)[0]) > 0.02:
        return None
    out = []
    prev, prev_a = float(x[near][-1]), float(a[near][-1])
    for xi, ai in zip(x[~near], a[~near]):
        if xi - prev > 12.0:
            break
        if abs(ai - prev_a) > 2e-3 * xi * (xi - prev):
            ref += ai - prev_a
        prev, prev_a = float(xi), float(ai)
        out.append((xi, ref - ai))
    return out


def classify(curv, i):
    lo, hi = max(0, i - 5), min(len(curv) - 1, i + 5)
    if abs(curv[hi] - curv[lo]) > 2e-4:
        return "entry/exit"
    return "curve" if abs(curv[i]) > 3e-4 else "straight"


stats = {}
for p in sys.argv[1:]:
    rows = [json.loads(l) for l in open(p)]
    curv = [r["curv"] for r in rows]
    for i, r in enumerate(rows):
        t = np.tan(r["yaw"])
        cls = classify(curv, i)
        best = {}
        for x, y, ok in r["sides"]:
            f = follow(x, y, r["c"], t, r["curv"])
            if not f:
                continue
            for xi, d in f:
                for b in BANDS:
                    if b[0] <= xi < b[1]:
                        if abs(d) > abs(best.get(b, 0.0)):
                            best[b] = d
        for b, d in best.items():
            stats.setdefault(cls, {}).setdefault(b, []).append(d)
for cls, bd in sorted(stats.items()):
    print(cls)
    for b in BANDS:
        v = np.array(bd.get(b, []))
        if v.size:
            print("   %3d-%3d m  n %5d  |err| median %.2f p90 %.2f  toward-axis > 0.8 m: %.1f%%" % (
                b[0], b[1], v.size, np.median(np.abs(v)), np.percentile(np.abs(v), 90), 100.0 * np.mean(v > 0.8)))
