"""Wall-drift contradiction range on gobs records (scratch). Variants by args.
usage: sig2.py <gobs.jsonl> [delta persist w0max gap mad] -> summary"""
import json, os, sys
INLIER = os.environ.get("INLIER") == "1"
import numpy as np


def drift_cap(x, y, c, t, k, delta=0.6, persist=2, w0max=3.5, gap=12.0, mad=0.25, near_max=30.0, min_w0=1.2,
              ret_w0=False, kmax=0.0, slope=0.0):
    if kmax > 0 or slope > 0:
        return drift_cap2(x, y, c, t, k, delta, persist, w0max, gap, mad, near_max, kmax, slope)
    """First X beyond near_max where the side's boundary has come delta nearer the axis than in the near
    field (median offset w0 of the bins within near_max, a side with w0 in [min_w0, w0max] and a near
    median absolute deviation <= mad), in `persist` consecutive populated bins, following the boundary while
    bins are no more than `gap` apart."""
    x = np.asarray(x, float); y = np.asarray(y, float)
    dy = y - (c + t * x + 0.5 * k * x * x)
    near = x <= near_max
    if near.sum() < 3:
        return None
    s = np.sign(np.median(dy[near]))
    a = s * dy
    w0 = float(np.median(a[near]))
    if not (min_w0 <= w0 <= w0max) or float(np.median(np.abs(a[near] - w0))) > mad:
        return None
    prev_x = float(x[near][-1])
    run, first = 0, None
    for xi, ai in zip(x[~near], a[~near]):
        if xi - prev_x > gap:
            break
        if ai < w0 - delta:
            run += 1
            if run == 1:
                first = float(xi)
            if run >= persist:
                return (first, w0) if ret_w0 else first
        else:
            run = 0
        prev_x = float(xi)
    return None


def drift_cap2(x, y, c, t, k, delta, persist, w0max, gap, mad, near_max, kmax, slope):
    """With steps: a bin-to-bin change larger than kmax * x * dx (a curvature error of kmax cannot move the
    boundary faster) is a structure edge: the reference moves with it. With slope: the near bins must have a
    line-fit slope of at most this (the boundary parallel to the rails)."""
    x = np.asarray(x, float); y = np.asarray(y, float)
    dy = y - (c + t * x + 0.5 * k * x * x)
    near = x <= near_max
    if near.sum() < 3:
        return None
    s = np.sign(np.median(dy[near]))
    a = s * dy
    w0 = float(np.median(a[near]))
    if INLIER:
        inl = near & (np.abs(a - w0) <= mad)
        if inl.sum() < 3:
            return None
        w0 = float(np.median(a[inl]))
        if not (0 < w0 <= w0max) or abs(np.polyfit(x[inl], a[inl], 1)[0]) > slope:
            return None
    else:
        if not (0 < w0 <= w0max) or float(np.median(np.abs(a[near] - w0))) > mad:
            return None
        if slope > 0:
            inl = near & (np.abs(a - w0) <= mad)
            if inl.sum() < 3 or abs(np.polyfit(x[inl], a[inl], 1)[0]) > slope:
                return None
    ref = w0
    prev_x, prev_a = float(x[near][-1]), float(a[near][-1])
    run, first = 0, None
    for xi, ai in zip(x[~near], a[~near]):
        if xi - prev_x > gap:
            break
        if kmax > 0 and abs(ai - prev_a) > kmax * xi * (xi - prev_x):
            ref += ai - prev_a
        prev_x, prev_a = float(xi), float(ai)
        if ai < ref - delta:
            run += 1
            if run == 1:
                first = float(xi)
            if run >= persist:
                return first - 2.0
        else:
            run = 0
    return None


def frame_cap(r, **kw):
    t = np.tan(r["yaw"])
    allsides = kw.pop("all", 0) if "all" in kw else 0
    caps = [drift_cap(x, y, r["c"], t, r["curv"], **kw) for x, y, ok in r["sides"] if ok or allsides]
    if allsides:
        kw["all"] = allsides
    caps = [v for v in caps if v is not None]
    return min(caps) if caps else None


NAMES = ["delta", "persist", "w0max", "gap", "mad", "kmax", "slope", "all"]


def parse(argv):
    kw = {}
    for n, v in zip(NAMES, argv):
        kw[n] = float(v) if n not in ("persist", "all") else int(v)
    return kw


if __name__ == "__main__":
    path = sys.argv[1]
    kw = parse(sys.argv[2:])
    rows = [json.loads(l) for l in open(path)]
    capped, loss, caps = 0, [], []
    for r in rows:
        cap = frame_cap(r, **kw)
        mon = r["mon"] or 0.0
        if cap is not None and cap < mon:
            capped += 1
            loss.append(mon - cap)
            caps.append(cap)
    n = len(rows)
    q = lambda a, p: round(float(np.percentile(a, p)), 1) if a else None
    lost = sum(loss)
    tot = sum(r["mon"] or 0 for r in rows)
    print("%-40s" % path.split("/")[-1], "frames", n, "capped", capped, "(%.1f%%)" % (100.0 * capped / max(n, 1)),
          "cap q10/50/90", q(caps, 10), q(caps, 50), q(caps, 90), "loss q50/90", q(loss, 50), q(loss, 90),
          "range lost %.1f%%" % (100 * lost / max(tot, 1)), "mon med", q([r["mon"] or 0 for r in rows], 50))
