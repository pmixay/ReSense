# wall-anchored adversarial ray-cast probes on C (and person+equipment pairs), sealed vs combined, one at a time

C="$J/roundT_pressureGate_roundT $ICP/icp_roundT_pressureGate_roundT.json"
cd $W/docs/evidence/judgement_2026-10-13
pr() { name=$1; shift
  [ -f $SR/out/wb_${name}_s.json ] || run_s $SR/wallbox_rt.py $C $SR/out/wb_${name}_s --start 0 --d0 230 "$@" > $SR/out/wb_${name}_s.log 2>&1

  tail -1 $SR/out/wb_${name}_s.log >> $SR/out/battery2.txt
  tail -1 $SR/out/wb_${name}_w.log >> $SR/out/battery2.txt
}
