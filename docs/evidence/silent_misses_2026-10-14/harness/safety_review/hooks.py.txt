"""Observation hooks + optional what-if perturbations for the safety review (scratch; no repo edits).
Import AFTER sys.path is set to the code root under test. Env:
  WT        code root; asserted against resense.__file__
  DARKBAND  deg (0 = off): before top_is_free, delete the frame returns of the cluster's OWN face above its top
            (azimuth span of the cluster +-0.02 deg, elevation top+0.04 .. top+DARKBAND, range within the
            top's tolerance + 0.5 m): a column whose face above the cluster top returns nothing (dark band /
            glass), the rays beside it unchanged. Only the free-top answer can change.
"""
import json, os, sys
import numpy as np
WT = os.environ["WT"]
import resense  # noqa
assert resense.__file__.startswith(WT), (resense.__file__, WT)
import resense.clustering as C

STATS = {"tf_calls": 0, "tf_free": 0, "tf_rows": [], "cut_parts": 0, "cut_rows": [], "k": -1}
DARK = float(os.environ.get("DARKBAND", "0") or 0)
_tf = getattr(C, "top_is_free", None)


def _dark(frame_xyz, idx, frame_ring):
    p = frame_xyz[idx].astype(np.float64)
    rh = np.hypot(p[:, 0], p[:, 1])
    el = np.degrees(np.arctan2(p[:, 2], rh))
    az = np.degrees(np.arctan2(p[:, 1], p[:, 0]))
    top = float(el.max())
    r_top = float(np.median(np.hypot(rh, p[:, 2])[el >= top - 0.06]))
    a0, a1 = float(az.min()) - 0.02, float(az.max()) + 0.02
    q = frame_xyz.astype(np.float64)
    qh = np.hypot(q[:, 0], q[:, 1])
    qe = np.degrees(np.arctan2(q[:, 2], qh))
    qa = np.degrees(np.arctan2(q[:, 1], q[:, 0]))
    qr = np.hypot(qh, q[:, 2])
    tol = max(0.5, 0.02 * r_top) + 0.5
    rm = (q[:, 0] > 0) & (qa >= a0) & (qa <= a1) & (qe > top + 0.04) & (qe <= top + DARK) & (np.abs(qr - r_top) <= tol)
    if not rm.any():
        return frame_xyz, idx, frame_ring, 0
    keep = ~rm
    newi = np.cumsum(keep) - 1
    return frame_xyz[keep], newi[idx], (None if frame_ring is None else frame_ring[keep]), int(rm.sum())


def top_is_free(frame_xyz, idx, min_returns=2, frame_ring=None):
    nrm = 0
    if DARK > 0 and idx.size >= 2:
        frame_xyz, idx, frame_ring, nrm = _dark(frame_xyz, idx, frame_ring)
    v = _tf(frame_xyz, idx, min_returns, frame_ring)
    STATS["tf_calls"] += 1
    STATS["tf_free"] += int(bool(v))
    if v or nrm:
        p = frame_xyz[idx]
        STATS["tf_rows"].append({"k": STATS["k"], "free": bool(v), "removed": nrm, "x": round(float(np.median(p[:, 0])), 1),
                                 "y": round(float(np.median(p[:, 1])), 2), "n": int(idx.size)})
    return v


if _tf is not None:
    C.top_is_free = top_is_free
if hasattr(C, "_off_structure_parts"):
    _cut = C._off_structure_parts

    def _off_structure_parts(b, xyz, h, in_gauge, vox, inv, cfg):
        parts = _cut(b, xyz, h, in_gauge, vox, inv, cfg)
        for p in parts:
            STATS["cut_parts"] += 1
            STATS["cut_rows"].append({"k": STATS["k"], "n_vox": int(p.n_vox), "x0": round(float(p.bmin[0]), 1),
                                      "size": [round(float(v), 2) for v in p.size],
                                      "y": round(float(np.median(p.pts[:, 1])), 2)})
        return parts
    C._off_structure_parts = _off_structure_parts


def dump(path):
    json.dump(STATS, open(path, "w"))
