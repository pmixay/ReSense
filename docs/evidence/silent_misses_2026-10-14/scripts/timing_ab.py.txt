#!/usr/bin/env python3
"""A/B timing (scratch): one code root's Detector over a slice of a gate cache, one process, fresh detector,
the cache stamps. usage: timing_ab.py <root> <cache_dir> <start> <count> <out.json>
Records Detector.process wall time and res.timing_ms["total"] per frame (warm-up: the first 20 frames dropped)."""
import json
import os
import sys
import time

ROOT = sys.argv[1]
sys.path.insert(0, ROOT)
import numpy as np  # noqa: E402

import resense  # noqa: E402

assert resense.__file__.startswith(ROOT + "/"), resense.__file__
from resense.config import DetectorConfig  # noqa: E402
from resense.detector import Detector  # noqa: E402
from resense.frame import frame_from_compact  # noqa: E402
from scripts.cache_io import cache_file_stem, cache_files, load_cache_array, load_cache_stamps  # noqa: E402

d, start, count, out = sys.argv[2], int(sys.argv[3]), int(sys.argv[4]), sys.argv[5]
files = cache_files(d)[start:start + count]
stamps = load_cache_stamps(d)
cfg = DetectorConfig.from_yaml(os.path.join(ROOT, "configs/default.yaml"))
det = Detector(cfg)
wall, tot = [], []
for j, f in enumerate(files):        # loaded one at a time (memory); only process() is timed
    stem = cache_file_stem(f)
    fr = frame_from_compact(load_cache_array(f), cfg.sensor, stamp=stamps.get(stem, j * 0.1), frame_id=stem)
    t = time.perf_counter()
    res = det.process(fr)
    wall.append((time.perf_counter() - t) * 1e3)
    tot.append(float(res.timing_ms["total"]))
w, t = np.array(wall[20:]), np.array(tot[20:])
from resense import _native  # noqa: E402
r = {"root": ROOT, "native": _native.status(),
     "frames": int(w.size), "start": start, "wall_mean": round(float(w.mean()), 2),
     "wall_p50": round(float(np.median(w)), 2), "wall_p95": round(float(np.percentile(w, 95)), 2),
     "total_mean": round(float(t.mean()), 2), "total_p95": round(float(np.percentile(t, 95)), 2)}
json.dump(r, open(out, "w"))
print("TIMING", json.dumps(r), flush=True)
