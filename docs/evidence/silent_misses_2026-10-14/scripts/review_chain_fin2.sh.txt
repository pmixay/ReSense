  [ -f $FR/out/wb_fin_$name.json ] || run_w $FR/wallbox_rt.py $bg $FR/out/wb_fin_$name "$@" > $FR/out/wb_fin_$name.log 2>&1
  python3 - $name >> $FR/out/wallbox_fin.txt <<'PY'
import json, sys
n = sys.argv[1]; SR = "/tmp/claude-0/fixwork/safety_review/out"; FR = "/tmp/claude-0/fixwork/fix_review/out"
def st(p):
    d = json.load(open(p)); o = d["summary"]["o0"]
    vis = [r for r in d["rows"] if r["objs"] and r["objs"][0]["n_obj"] > 0]
    dec = {k: sum(r["decision"] == k for r in vis) for k in ("GO", "CAUTION", "STOP")}
    return o["obj_stop_frames"], o["first_stop_X"], dec, d["summary"]["decisions"], d["summary"]["hooks"].get("cut_parts")
s, w, f = st(f"{SR}/wb_{n}_s.json"), st(f"{SR}/wb_{n}_w.json"), st(f"{FR}/wb_fin_{n}.json")
print(f"{n:16s} objSTOP sealed {s[0]} (first {s[1]}) 13.10 {w[0]} (first {w[1]}) fix {f[0]} (first {f[1]}) | while visible sealed {s[2]} fix {f[2]} | all frames sealed {s[3]} fix {f[3]} | cut parts 13.10 {w[4]} fix {f[4]}")
PY
}
