import json, re
def parse(path, pat):
    rows = {}; last = None
    for l in open(path).read().splitlines():
        m = re.match(r"^(plain|rob)?\s*(\S+) (judge-baseline|sealed|13\.10)->(?:fin|\S+) frames (\d+) STOP (\d+) -> (\d+) episodes (\d+) -> (\d+) CAUTION (\d+) -> (\d+) decision diffs (\d+) det/warn diffs (\d+)", l)
        if m:
            n = m.group(2); base = m.group(3)
            r = rows.setdefault(n, {"frames": int(m.group(4))})
            key = "13.10" if base == "13.10" else "sealed"
            r[key] = {"STOP": int(m.group(5)), "episodes": int(m.group(7)), "CAUTION": int(m.group(9)), "base_is": base}
            r["now" if key == "sealed" else "_now13"] = {"STOP": int(m.group(6)), "episodes": int(m.group(8)), "CAUTION": int(m.group(10))}
            r[f"decision_diffs_vs_{key}"] = int(m.group(11)); r[f"detail_diffs_vs_{key}"] = int(m.group(12))
            last = n
        m = re.match(r"^out/(?:plain_|rob_)?(\S+)_fin frames (\d+) tf (\d+) free (\d+) cut (\d+)", l)
        if m and last:
            rows[last].update(tf_calls=int(m.group(3)), tf_free=int(m.group(4)), cut_parts=int(m.group(5)))
    for r in rows.values():
        r.pop("_now13", None)
    return rows
