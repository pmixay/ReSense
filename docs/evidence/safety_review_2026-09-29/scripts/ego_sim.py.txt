"""Tracker-level simulations of tracking.ego_veto_* (default TrackingConfig except the learned doubt model off)."""
from dataclasses import replace
import numpy as np
from resense.clustering import Cluster
from resense.config import TrackingConfig
from resense.tracking import Tracker

def cl(x, zone="gauge", reason="", ng=40):
    c = np.array([x, 0.0, 1.0])
    return Cluster(points_idx=np.arange(3), n=40, n_raw=40, centroid=c, bbox_min=c - 0.2, bbox_max=c + 0.2,
                   distance=x, lateral=0.0, height_min=0.2, height_max=1.7, intensity=10.0, n_expected=40.0,
                   score=1.0, zone=zone, n_gauge=ng, reason=reason)

def cfg(**kw):
    base = dict(doubt_extra_hits=0)
    base.update(kw)
    return TrackingConfig(**base)

def run(dists, est, true_v, c, hitmask=None):
    """dists: object distance per frame; est: odo speed per frame (None ok). Returns first STOP (frame, distance)."""
    t = Tracker(c)
    for k, d in enumerate(dists):
        clusters = [cl(d)] if (hitmask is None or hitmask[k]) else []
        t.update(clusters, ego_shift=0.0, frame_dt=0.1, far_thin=[], odo_speed=est[k])
        for tr in t.tracks:
            if tr.reported and tr.zone == "gauge":
                return k, round(d, 1)
    return None

on, off = cfg(), cfg(ego_veto_min_speed=0.0)
print("A. static object from 150 m, true speed v, estimate = k*v (clipped 30): first STOP distance on / off")
for v in (4, 6, 8, 10, 15, 22):
    row = []
    for k in (1.0, 2.0, 2.5, 3.0, 4.0):
        e = min(30.0, k * v)
        n = int((150 - 3) / (v * 0.1))
        dists = [150 - v * 0.1 * i for i in range(n)]
        a = run(dists, [e] * n, v, on); b = run(dists, [e] * n, v, off)
        row.append(f"k={k}: {a[1] if a else None}/{b[1] if b else None}")
    print(f"  v={v:>4}: " + "  ".join(row))

print("B. object moving away at u, train v (estimate exact), start 100 m ahead: first STOP distance on / off")
for v in (4, 6, 8, 12, 22):
    row = []
    for u in (1.5, 2.5, 3.5, 5.0, 8.0):
        if u >= v: row.append(f"u={u}: -"); continue
        n = int((100 - 3) / ((v - u) * 0.1)); dists = [100 - (v - u) * 0.1 * i for i in range(n)]
        a = run(dists, [v] * n, v, on); b = run(dists, [v] * n, v, off)
        row.append(f"u={u}: {a[1] if a else None}/{b[1] if b else None}")
    print(f"  v={v:>4}: " + "  ".join(row))

print("C. person running away at u then stopping (falls) at distance D, train v: STOP after the stop, frames (on / off)")
for v, u, D in ((6, 4.5, 80), (6, 4.5, 40), (8, 6, 60), (10, 7, 60), (5, 4, 45)):
    dists = []; d = 120.0; stopped_at = None
    while d > 3:
        dists.append(d)
        if d > D: d -= (v - u) * 0.1
        else:
            if stopped_at is None: stopped_at = len(dists) - 1
            d -= v * 0.1
    a = run(dists, [v] * len(dists), v, on); b = run(dists, [v] * len(dists), v, off)
    print(f"  v={v} u={u} stops at {D} m (frame {stopped_at}): on {a}  off {b}")

print("D. static object, true v = est = 22 (and 10), Gaussian distance noise sigma: share of 400 runs whose first STOP is later with the veto, and worst delay")
rng = np.random.default_rng(1)
for v in (22.0, 10.0):
    for s in (0.3, 0.5, 1.0, 2.0):
        later = 0; worst = 0; nruns = 400
        for r in range(nruns):
            n = int((150 - 3) / (v * 0.1))
            dists = [150 - v * 0.1 * i + rng.normal(0, s) for i in range(n)]
            a = run(dists, [v] * n, v, on); b = run(dists, [v] * n, v, off)
            if a != b:
                later += 1; worst = max(worst, (a[0] if a else n) - b[0])
        print(f"  v={v} sigma={s}: delayed in {later}/{nruns}, worst delay {worst} frames ({worst * v * 0.1:.1f} m)")

print("E. train standing (true 0) but the speed input says s (spurious estimate or a constant ego_speed_mps): static object at 60 m")
for s in (None, 0.0, 3.9, 4.0, 6.0, 22.0):
    n = 60; dists = [60.0] * n
    a = run(dists, [s] * n, 0, on); b = run(dists, [s] * n, 0, off)
    print(f"  speed input {s}: on {a}  off {b}")

print("F. true v, speed input constant 22 (ego_speed_mps:=22): static object from 150 m")
for v in (3, 5, 7, 7.7, 8, 10):
    n = int((150 - 3) / (v * 0.1)); dists = [150 - v * 0.1 * i for i in range(n)]
    a = run(dists, [22.0] * n, v, on); b = run(dists, [22.0] * n, v, off)
    print(f"  true v={v}: on {a}  off {b}")

print("G. sparse far returns: static object at 22 m/s, hit prob p (ego veto only)")
for p in (0.6, 0.8):
    later = 0
    for r in range(300):
        n = int((150 - 3) / 2.2); dists = [150 - 2.2 * i for i in range(n)]
        hm = rng.random(n) < p
        a = run(dists, [22.0] * n, 22, on, hm); b = run(dists, [22.0] * n, 22, off, hm)
        later += a != b
    print(f"  p={p}: first STOP differs in {later}/300 runs")
